function setup() {
  createCanvas(800, 600);
  noLoop();
}

function draw() {
  background(220);
  for(y=0;y<height;++y){
    for(x=0;x<width;++x){
      set(x,y, color((x+y)/(width+height)*255, sqrt( pow((width/2)-x,2)+pow((height/2)-y,2) ),255-sqrt( pow((width/2)-x,2)+pow((height/2)-y,2) ) )) 
    }    
  }
  updatePixels();
}
