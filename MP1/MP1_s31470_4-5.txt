function setup() {
  createCanvas(800, 600);
  noLoop();
}

function draw() {
  background(200,200,251);
  for(y=400;y<height;++y)
       for(x=0;x<width;++x){
         set(x,y,color(41,98,24))
  }
  //flowers 
  for(n=0;n<1000;n++){
    set(floor(random(0,800)),floor(random(400,600)), color(floor(random(0,255)),floor(random(0,255)),floor(random(0,255))))
  }
  for(y=200;y<400;++y){
    for(x=200;x<600;++x){
      set(x,y,color(93,53,34))
    }
  }
  //roof
  for(y=20, r=0;y<200;++y,r+=2)
    for(x=0;x<800;++x){
      if(x<width/2-r || x>width/2+r ){
      } else{
        set(x,y,color(220,117,109))
      }
    }
    
  updatePixels();
}
