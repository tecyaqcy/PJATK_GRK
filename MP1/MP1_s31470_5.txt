function setup() {
  createCanvas(800, 600);
  noLoop();
}

function draw() {
  background(0);
  x1 = 400, y1 = 100;
  x2 = 200, y2 = 500;
  x3 = 600, y3 = 500;
  set(x1, y1, color(255));
  set(x2, y2, color(255));
  set(x3, y3, color(255));
  cx = x1, cy = y1
  for(y=0; y<30000; y++){
    temp = floor(random(0,3));
    if(temp==0){
      cx = (cx+x1)/2;
      cy = (cy+y1)/2;
      set(cx,cy,color(255));
    } else if(temp==1){
      cx = (cx+x2)/2;
      cy = (cy+y2)/2;
      set(cx,cy,color(255));
    } else{
      cx = (cx+x3)/2;
      cy = (cy+y3)/2;
      set(cx,cy,color(255));
    }
  }
 
  updatePixels();
}
